% Crossed product: M (type III_1, with its modular flow sigma_s) together with
% a clock (a particle on a line, Hilbert space L^2(R)) gives \widehat M (type II_infty).
% Included at 0.88\textwidth (5.72in); drawn at ~1:1.
\documentclass[tikz,border=3pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{xcolor}
\usetikzlibrary{arrows.meta}

\definecolor{boxred}{HTML}{fbecec}
\definecolor{boxblue}{HTML}{e8f0fa}
\definecolor{boxgreen}{HTML}{e6f4ed}
\definecolor{ruleblue}{HTML}{2b6cb0}
\definecolor{rulered}{HTML}{c0392b}
\definecolor{rulegreen}{HTML}{1f8a5f}
\definecolor{axisgray}{HTML}{8a8f98}

\newcommand{\M}{\mathcal{M}}

\begin{document}
\begin{tikzpicture}[>=Latex, scale=1.27]
  % ---- M with its modular flow ----
  \fill[boxred] (0,0) circle (1.0);
  \draw[rulered, line width=0.8pt] (0,0) circle (1.0);
  \node[font=\large] at (0,0) {$\M$};
  \draw[-{Latex[length=2.2mm]}, rulered, line width=0.9pt]
    (125:1.32) arc[start angle=125, end angle=55, radius=1.32];
  \node[rulered] at (90:1.58) {$\sigma_s$};
  \node[rulered] at (0,-1.3) {III$_1$};

  % ---- plus ----
  \node[font=\Large, axisgray!70!black] at (1.85,0) {$+$};

  % ---- clock: a particle on a line with a wavepacket ----
  \begin{scope}[xshift=4.1cm]
    \fill[boxblue, rounded corners=6pt] (-1.55,-1.0) rectangle (1.55,1.0);
    \draw[ruleblue, line width=0.8pt, rounded corners=6pt] (-1.55,-1.0) rectangle (1.55,1.0);
    \draw[->, line width=0.5pt] (-1.25,-0.45) -- (1.3,-0.45) node[above left=-1pt, font=\small] {$\hat q$};
    \fill[ruleblue!25] plot[domain=-1.2:1.2, samples=80, variable=\u]
      ({\u}, {-0.45 + 1.05*exp(-(\u+0.15)^2/0.12)}) -- (1.2,-0.45) -- (-1.2,-0.45) -- cycle;
    \draw[ruleblue, line width=0.9pt, domain=-1.2:1.2, samples=80, variable=\u]
      plot ({\u}, {-0.45 + 1.05*exp(-(\u+0.15)^2/0.12)});
    \node[ruleblue] at (0,-1.3) {$L^2(\mathbb{R})$};
  \end{scope}

  % ---- arrow ----
  \draw[-{Latex[length=3mm]}, line width=1pt] (6.05,0) -- (7.45,0);

  % ---- crossed product ----
  \begin{scope}[xshift=8.85cm]
    \fill[boxgreen] (0,0) circle (1.1);
    \draw[rulegreen, line width=0.8pt] (0,0) circle (1.1);
    \node[font=\large] at (0,0) {$\widehat\M$};
    \node[rulegreen] at (0,-1.3) {II$_\infty$};
  \end{scope}
\end{tikzpicture}
\end{document}
