\begin{tikzpicture}[font=\small, line cap=round, line join=round]
  % ---------- (a) Page curve ----------
  % S_BH(t) = S0 (1 - t/te)^{2/3};  S_Hawking(t) = c (S0 - S_BH), c = 1.5
  % Page time: (1 - tP/te)^{2/3} = c/(1+c) = 0.6  ->  tP/te = 1 - 0.6^{1.5}
  \def\te{6.2}  \def\So{2.7}  \def\c{1.5}
  \pgfmathsetmacro{\uP}{1-0.6^1.5}
  \pgfmathsetmacro{\xP}{\te*\uP}
  \pgfmathsetmacro{\yP}{0.6*\So}
  \draw[-{Latex[length=4pt]}, thin] (0,0) -- (\te+0.5,0) node[below] {$t$};
  \draw[-{Latex[length=4pt]}, thin] (0,0) -- (0,4.4) node[left] {$S(R)$};
  % other branches
  \draw[cred, dashed, thin, domain=\uP:1, samples=60]
    plot ({\te*\x}, {\c*\So*(1-(1-\x)^(2/3))});
  \draw[cblue, dashed, thin, domain=0:\uP, samples=60]
    plot ({\te*\x}, {\So*(1-\x)^(2/3)});
  % Page curve = minimum
  \draw[cgreen, line width=1.4pt, domain=0:\uP, samples=60]
    plot ({\te*\x}, {\c*\So*(1-(1-\x)^(2/3))});
  % after t_P: write 1 - t/te = s^3, so S = So s^2 stays smooth down to s = 0
  \pgfmathsetmacro{\sP}{0.6^0.5}
  \draw[cgreen, line width=1.4pt, domain=\sP:0, samples=120, variable=\s]
    plot ({\te*(1-\s^3)}, {\So*\s^2});
  \draw[cgray, dotted, thin] (\xP,0) -- (\xP,\yP);
  \node[below] at (\xP,0) {$t_P$};
  \node at (\te/2,-1.0) {(a)};

  % ---------- (b) Penrose diagram of an evaporating black hole with an island ----------
  % Complete diagram, laid out after the island figure of Almheiri, Hartman, Maldacena,
  % Shaghoulian and Tajdini, Rev. Mod. Phys. 93, 035002 (2021).  Conformal coordinates,
  % null rays at 45 degrees:
  %   i^- = (0,0), i^0 = (5,5), i^+ = (2.4,7.6);  I^-: y = x;  I^+: y = 10 - x;
  %   r = 0: x = 0 for 0 < y < 6.2, and x = 2.4 for 6.2 < y < 7.6 (after the evaporation);
  %   singularity y = 6.2 from (0,6.2) to the endpoint E = (2.4,6.2);
  %   event horizon y = x + 3.8, from (0,3.8) to E.
  % Checked numerically: the star surface and the cutoff surface are timelike everywhere, the
  % cutoff stays outside the black hole and inside the diagram, all three slice segments are
  % spacelike, and the QES lies just inside the horizon, a short advanced time before the
  % point where the slice meets the cutoff.
  \begin{scope}[shift={(8.2,-0.5)}, scale=0.72]
    % collapsing star
    \fill[tgold] (0,0) .. controls (0.9,1.3) and (0.9,3.0) .. (0.05,6.2) -- (0,6.2) -- cycle;
    \draw[cgold, thin] (0,0) .. controls (0.9,1.3) and (0.9,3.0) .. (0.05,6.2);
    % r = 0, before and after the evaporation
    \draw[thin] (0,0) -- (0,6.2);
    \draw[thin] (2.4,6.2) -- (2.4,7.6);
    % singularity and event horizon
    \draw[decorate, decoration={zigzag, segment length=3.2pt, amplitude=1.1pt}, thick]
      (0,6.2) -- (2.4,6.2);
    \draw[cred, dashed, thin] (0,3.8) -- (2.4,6.2);
    % past and future null infinity
    \draw[thin] (0,0) -- (5,5) -- (2.4,7.6);
    % cutoff surface separating the black-hole region from the radiation region
    \draw[cgray, densely dotted, line width=0.9pt]
      (0,0) .. controls (2.3,2.7) and (3.9,5.3) .. (2.4,7.6);
    % Cauchy slice after the Page time
    \coordinate (p0) at (0,5.2);
    \coordinate (q)  at (1.613,5.553);
    \coordinate (c)  at (2.864,4.701);
    \coordinate (i0) at (5,5);
    \draw[cgreen, line width=1.4pt] (p0) .. controls (0.6,5.25) and (1.1,5.36) .. (q);
    \draw[black!70, line width=0.9pt] (q) .. controls (2.0,5.33) and (2.45,4.98) .. (c);
    \draw[cblue, line width=1.4pt] (c) .. controls (3.6,4.55) and (4.4,4.72) .. (i0);
    \fill[cred] (q) circle (2.4pt);
    \node[cgreen, above] at (0.75,5.28) {$I$};
    \node[below left] at (2.3,5.1) {$O$};
    \node[cblue, below] at (3.95,4.62) {$R$};
  \end{scope}
  \node at (8.2+0.72*2.5,-1.0) {(b)};
\end{tikzpicture}
